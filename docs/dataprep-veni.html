8 Inspecting duplicates

This is not the case in this curated example, but it did occur a few times (<5) in our data, so we show how we dealt with duplicate Veni matches here.

duplicatedids <- phds$id[which(duplicated(phds$id))]


inspect <- phds[phds$id %in% duplicatedids, ]

# there is 1 duplicate, so 2 observations to expect for this duplicate, one of the matches leads to
# the same year of veni receipt, and the other diverges. So the match where both indicate the same
# year of Veni receipt is likely the true match

inspect$choose <- as.numeric(inspect$year1 == inspect$year2)

inspect <- inspect %>%
    filter(choose == 1)  # remove the cases where the veni years diverge
inspect <- subset(inspect, select = -choose)

# remove the duplicates (all 6) from the phd data
phds <- phds %>%
    filter(!phds$id %in% duplicatedids)

# add back only the correct
phds <- rbind.data.frame(phds, inspect)
