fpackage.check <- function(packages) {
  lapply(packages, FUN = function(x) {
    if (!require(x, character.only = TRUE)) {
      install.packages(x, dependencies = TRUE)
      library(x, character.only = TRUE)
    }
  })
}

bootFunc <- function(data, i) {
  df <- data[i,] #bootstrap datasets
  
  M1 <- glm(formula = veni ~ as.factor(gender_2) + as.factor(time), family = binomial(link="cloglog"), data = df)
  
  M2 <- glm(formula = veni ~ as.factor(gender_2) * phd_year + as.factor(time), family = binomial(link="cloglog"), data = df)
  
  M3a <- glm(formula = veni ~ phd_year * as.factor(gender_2) + as.factor(time) + npubs_cumul + npubs_prev + field + uni, family = binomial(link="cloglog"), data = df)
  
  M3b <- glm(formula = veni ~ phd_year * as.factor(gender_2) + as.factor(time) + npubs_cumul + npubs_prev + field + as.factor(uni) + teamsize + as.factor(prof2p) + npubs_sv_norm, family = binomial(link="cloglog"), data = df)
  
  M3c <- glm(formula = veni ~ phd_year * as.factor(gender_2) + as.factor(time) + npubs_cumul + npubs_prev + field + uni + teamsize + as.factor(prof2p)  + npubs_sv_norm + as.factor(wom_sv1), family = binomial(link="cloglog"), data = df)
  
  M4 <- glm(formula = veni ~ phd_year * as.factor(gender_2) + as.factor(time) + npubs_cumul + npubs_prev + field + uni + teamsize + as.factor(prof2p)  + npubs_sv_norm + as.factor(wom_sv1)*as.factor(gender_2), family = binomial(link="cloglog"), data = df)
  

    suppressWarnings({
  
    # CREATING THE DATASETS
    
    s <- .1
    
    # gender datasets  
    dfmen <- dfwomen <- df
    dfwomen$gender_2 <- "women" # one dataset of all women
    dfmen$gender_2 <- "men" # one dataset of all men
    
    dfwomen$gender_2 <- factor(dfwomen$gender_2, levels=levels(df$gender_2))
    dfmen$gender_2 <- factor(dfmen$gender_2, levels=levels(df$gender_2))
    
    # time datasets
    time_levels <- levels(as.factor(df$time))
    ref_time <- "0"
    
    # cohort * gender datasets
    dfwcp <- dfwcm <- dfmcp <- dfmcm <- df
    dfwcp$gender_2 <- dfwcm$gender_2 <- "women"
    dfmcp$gender_2 <- dfmcm$gender_2 <- "men"
    
    dfwcp$gender_2 <- factor(dfwcp$gender_2, levels=levels(df$gender_2))
    dfwcm$gender_2 <- factor(dfwcm$gender_2, levels=levels(df$gender_2))
    dfmcp$gender_2 <- factor(dfmcp$gender_2, levels=levels(df$gender_2))
    dfmcm$gender_2 <- factor(dfmcm$gender_2, levels=levels(df$gender_2))
    
    dfwcp$phd_year <- dfmcp$phd_year <- df$phd_year + s
    dfwcm$phd_year <- dfmcm$phd_year <- df$phd_year - s
    
    # cohort datasets
    dfcp <- dfcm <- df
    dfcp$phd_year <- df$phd_year + s
    dfcm$phd_year <- df$phd_year - s
    
    # cumulative pubs
    dfpcp <- dfpcm <- df
    dfpcp$npubs_cumul <- df$npubs_cumul + s
    dfpcm$npubs_cumul <- df$npubs_cumul - s
    
    # previous pubs
    dfppp <- dfppm <- df
    dfppp$npubs_prev <- df$npubs_prev + s
    dfppm$npubs_prev <- df$npubs_prev - s
    
    # network size datasets
    dfnsp <- dfnsm <- df
    dfnsp$teamsize <- dfnsp$teamsize + s
    dfnsm$teamsize <- dfnsm$teamsize - s
    
    # professor datasets
    dfprof <- dfnoprof <- df
    dfprof$prof2p <- 1 
    dfnoprof$prof2p <- 0
    
     # pubs supervisor datasets
    dfspp <- dfspm <- df
    dfspp$npubs_sv_norm <- dfspp$npubs_sv_norm + s
    dfspm$npubs_sv_norm <- dfspm$npubs_sv_norm - s
    
    # women sv datasets
